    <h2>График: $\sin(x) \ge 2x^2 - 1$</h2>
    <canvas id="mathChart"></canvas>
</div>

<script>
    // Генерируем точки для оси X от -1 до 1 с шагом 0.05
    const xValues = [];
    const ySin = [];
    const yParabola = [];

    for (let x = -1.0; x <= 1.0; x += 0.05) {
        xValues.push(Number(x.toFixed(2)));
        ySin.push(Math.sin(x));
        yParabola.push(2 * x * x - 1);
    }

    const data = {
        labels: xValues,
        datasets: [
            {
                label: 'y = sin(x)',
                data: ySin,
                borderColor: 'blue',
                borderWidth: 2,
                fill: false,
                pointRadius: 0
            },
            {
                label: 'y = 2x² - 1',
                data: yParabola,
                borderColor: 'red',
                borderWidth: 2,
                fill: false,
                pointRadius: 0
            }
        ]
    };

    const config = {
        type: 'line',
        data: data,
        options: {
            responsive: true,
            plugins: {
                legend: {
                    position: 'top',
                }
            },
            scales: {
                x: {
                    title: {
                        display: true,
                        text: 'x (ОДЗ: [-1, 1])'
                    }
                },
                y: {
                    title: {
                        display: true,
                        text: 'y'
                    },
                    min: -1.2,
                    max: 1.2
                }
            }
        }
    };

    const mathChart = new Chart(
        document.getElementById('mathChart'),
        config
    );
</script>
